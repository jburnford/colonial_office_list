================================================================================
OUTPUT FORMAT
================================================================================

Returns dictionary matching FileAnalysis dataclass from extract_people_v2.py:

{
  "file_path": str,
  "colony": str,
  "year": int,
  "people_section_start": int,      # Start line number
  "people_section_end": int,        # End line number
  "start_marker": str,              # Text marking the start
  "departments": List[str],         # List of departments
  "provinces": List[str],           # List of provinces
  "primary_format": str,            # Format description
  "has_lists": bool,                # Comma-separated lists?
  "has_ditto": bool,                # Ditto marks present?
  "salary_currency": str,           # Currency format
  "ocr_quality": str,               # good/fair/poor/unknown
  "extraction_notes": List[str]     # Additional notes
}

   from file_analyzer_llm import analyze_file_structure

               year=year
           )
           return FileAnalysis(**analysis_dict)
